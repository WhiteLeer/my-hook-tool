cmake_minimum_required(VERSION 3.19)
project(GuestFactoryProbe LANGUAGES C CXX ASM)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
  message(FATAL_ERROR "The passive entry probe requires Android arm64-v8a")
endif()
set(DOBBY_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/dobby" CACHE PATH "Dobby source directory")
if(NOT EXISTS "${DOBBY_SOURCE}/include/dobby.h")
  message(FATAL_ERROR "Pass -DDOBBY_SOURCE=<Dobby checkout>")
endif()
set(NearBranch ON CACHE BOOL "Near-only entry experiment" FORCE)
set(FullFloatingPointRegisterPack ON CACHE BOOL "Preserve complete vector register context" FORCE)
set(DOBBY_DEBUG OFF CACHE BOOL "" FORCE)
add_subdirectory("${DOBBY_SOURCE}" dobby EXCLUDE_FROM_ALL)
target_compile_options(dobby_static PRIVATE -include limits.h -DLINE_MAX=4096)
target_compile_definitions(dobby_static PRIVATE MYHOOK_STRICT_ENTRY_ONLY=1)
add_library(GuestFactoryProbe SHARED GuestFactoryProbe.cpp)
target_compile_features(GuestFactoryProbe PRIVATE cxx_std_20)
target_link_libraries(GuestFactoryProbe PRIVATE dobby_static log dl)
target_include_directories(GuestFactoryProbe PRIVATE "${DOBBY_SOURCE}/include")
target_compile_options(GuestFactoryProbe PRIVATE -Wall -Wextra -Werror)
target_link_options(GuestFactoryProbe PRIVATE "-Wl,--build-id=sha1" "-Wl,-Bsymbolic-functions")
